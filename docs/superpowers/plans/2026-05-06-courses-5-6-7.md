import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
from utils import get_api_key

from langchain_community.document_loaders import TextLoader
from langchain_community.embeddings import DashScopeEmbeddings
from langchain_community.vectorstores import Redis
from langchain_classic.text_splitter import CharacterTextSplitter
from langchain_text_splitters import RecursiveCharacterTextSplitter

DOCUMENT_FILE = Path(__file__).resolve().parent.parent / "第1课_最小RAG" / "alibaba.txt"
DASHSCOPE_API_KEY = get_api_key()
TEST_QUESTION = "阿里巴巴的主要业务有哪些？"

embeddings = DashScopeEmbeddings(
    model="text-embedding-v3",
    dashscope_api_key=DASHSCOPE_API_KEY,
)


def load_docs():
    loader = TextLoader(str(DOCUMENT_FILE), encoding="utf-8")
    return loader.load()


def build_store(splitter, index_name: str):
    docs = load_docs()
    chunks = splitter.split_documents(docs)
    store = Redis.from_documents(
        documents=chunks,
        embedding=embeddings,
        redis_url="redis://localhost:6379",
        index_name=index_name,
    )
    return store, chunks


def show_results(store, k: int, label: str):
    retriever = store.as_retriever(search_kwargs={"k": k})
    results = retriever.invoke(TEST_QUESTION)
    print(f"\n  {label} | k={k} | 检索到 {len(results)} 条")
    for i, doc in enumerate(results, 1):
        preview = doc.page_content[:80].replace("\n", " ")
        print(f"  [{i}] ({len(doc.page_content)}字) {preview}...")


# ────────────────────────────────────────────
# Demo 1: Splitter 类型对比
# CharacterTextSplitter 按固定字符切；RecursiveCharacterTextSplitter 优先按段落/句子切
# ────────────────────────────────────────────
def demo_splitter_comparison():
    print("\n" + "=" * 55)
    print("Demo 1: Splitter 类型对比")
    print(f"问题: {TEST_QUESTION}")
    print("=" * 55)

    char_splitter = CharacterTextSplitter(chunk_size=500, chunk_overlap=100, separator="\n")
    char_store, char_chunks = build_store(char_splitter, "rag5_char")
    print(f"\nCharacterTextSplitter: 切出 {len(char_chunks)} 个 chunk")
    show_results(char_store, k=3, label="CharacterTextSplitter")

    rec_splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=100)
    rec_store, rec_chunks = build_store(rec_splitter, "rag5_recursive")
    print(f"\nRecursiveCharacterTextSplitter: 切出 {len(rec_chunks)} 个 chunk")
    show_results(rec_store, k=3, label="RecursiveCharacterTextSplitter")


# ────────────────────────────────────────────
# Demo 2: chunk_size 调优
# 同一文档，chunk_size 越小切得越碎，越大信息越完整但噪音也越多
# ────────────────────────────────────────────
def demo_chunk_size():
    print("\n" + "=" * 55)
    print("Demo 2: chunk_size 调优")
    print(f"问题: {TEST_QUESTION}")
    print("=" * 55)

    for chunk_size, overlap in [(200, 40), (500, 100), (1000, 200)]:
        splitter = RecursiveCharacterTextSplitter(chunk_size=chunk_size, chunk_overlap=overlap)
        store, chunks = build_store(splitter, f"rag5_size{chunk_size}")
        print(f"\nchunk_size={chunk_size}, overlap={overlap}: 共 {len(chunks)} 个 chunk")
        show_results(store, k=3, label=f"size={chunk_size}")


# ────────────────────────────────────────────
# Demo 3: k 值调整
# k 越大召回越多，但噪音也越多；k 越小精度高，但可能漏掉关键内容
# ────────────────────────────────────────────
def demo_k_value():
    print("\n" + "=" * 55)
    print("Demo 3: k 值调整（chunk_size 固定 500）")
    print(f"问题: {TEST_QUESTION}")
    print("=" * 55)

    splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=100)
    store, _ = build_store(splitter, "rag5_ktest")
    for k in [1, 3, 5]:
        show_results(store, k=k, label=f"k={k}")


if __name__ == "__main__":
    demo_splitter_comparison()
    demo_chunk_size()
    demo_k_value()
```

# 第5课：RAG 质量调优

## 这节课解决什么问题

第1课实现了最小 RAG，但效果可能不稳定：有时检索结果和问题毫无关系，有时找到的内容不完整。

问题出在这几个地方：
- **切分方式**：切得太碎丢失上下文，切得太大引入噪音
- **chunk_size**：块大小直接影响每个向量的语义密度
- **k 值**：检索数量太少漏掉答案，太多引入干扰

---

## 核心概念：为什么切分方式这么重要

RAG 的核心是：把文档切成小块 → 向量化 → 检索最相关的块。

切块的好坏直接影响检索质量：

```
原文：
"阿里巴巴成立于1999年。公司旗下有淘宝、天猫、菜鸟等业务。
 其中淘宝是最大的C2C电商平台，天猫是B2C平台。"

CharacterTextSplitter（按字符数强制切）可能切成：
  块1: "阿里巴巴成立于1999年。公司旗下有淘宝、天猫、"   ← 句子被截断
  块2: "菜鸟等业务。其中淘宝是最大的C2C电商平台，"      ← 语义不完整

RecursiveCharacterTextSplitter（按段落/句子递归切）：
  块1: "阿里巴巴成立于1999年。公司旗下有淘宝、天猫、菜鸟等业务。"
  块2: "其中淘宝是最大的C2C电商平台，天猫是B2C平台。"  ← 语义完整
```

---

## 三个可调参数

### 参数 1：Splitter 类型

| Splitter | 切法 | 适合场景 |
|---|---|---|
| `CharacterTextSplitter` | 按固定字符数，遇到 `separator` 就切 | 格式规整的文本（代码、日志） |
| `RecursiveCharacterTextSplitter` | 优先按段落 → 句子 → 词 递归切，尽量保持语义完整 | **通用首选**，适合自然语言文档 |

```python
from langchain_classic.text_splitter import CharacterTextSplitter
from langchain_text_splitters import RecursiveCharacterTextSplitter

# 强制按换行符切，不保证语义完整
char_splitter = CharacterTextSplitter(chunk_size=500, chunk_overlap=100, separator="\n")

# 优先按段落切，退而求其次按句子切（推荐）
rec_splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=100)
```

### 参数 2：chunk_size 和 chunk_overlap

```python
RecursiveCharacterTextSplitter(
    chunk_size=500,    # 每块最大字符数，越小越碎，越大越完整
    chunk_overlap=100, # 块与块之间重叠字符数，防止答案跨块时被截断
)
```

| chunk_size | 优点 | 缺点 |
|---|---|---|
| 200（小） | 向量语义集中，精确 | 容易截断长句，信息不完整 |
| 500（中） | 平衡语义完整与精确性 | **通常的起点** |
| 1000（大） | 保留更多上下文 | 向量语义稀释，检索精度下降 |

**chunk_overlap 经验值**：设为 chunk_size 的 10-20%（如 size=500 时 overlap=50-100）

### 参数 3：k 值（检索数量）

```python
retriever = store.as_retriever(search_kwargs={"k": 3})
```

| k | 效果 |
|---|---|
| k=1 | 精确但风险大，答案可能就在第2条 |
| k=3 | **通常的默认值**，平衡精度和覆盖率 |
| k=5 | 召回率高，但给 LLM 的上下文更长，成本更高 |

---

## 调优流程

```
1. 先用 RecursiveCharacterTextSplitter + chunk_size=500 + k=3 跑通
2. 肉眼看检索结果：
   - 结果和问题无关 → 可能 chunk_size 太大，试试 200-300
   - 结果内容不完整 → 增大 chunk_size 或增大 k
   - 结果重复内容多 → 减小 chunk_overlap
3. 固定 Splitter 和 chunk_size 后，再微调 k 值
```

import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
from utils import get_llm

from langchain.agents import create_react_agent, AgentExecutor
from langchain_core.tools import tool
from langchain_core.prompts import PromptTemplate

llm = get_llm()


# ────────────────────────────────────────────
# 工具定义（全部 mock，无需真实 API）
# ────────────────────────────────────────────
@tool
def search_weather(city: str) -> str:
    """查询指定城市的今日天气，支持北京、上海、广州"""
    mock_data = {
        "北京": "晴天，气温 22°C，湿度 40%，微风",
        "上海": "多云，气温 28°C，湿度 75%，东南风3级",
        "广州": "阵雨，气温 30°C，湿度 90%，南风2级",
    }
    return mock_data.get(city, f"{city}：暂无数据（支持：北京、上海、广州）")


@tool
def calculate(expr: str) -> str:
    """计算数学表达式，例如 '100 * 7.25' 或 '22 + 100'"""
    try:
        result = eval(expr, {"__builtins__": {}}, {})
        return f"{expr} = {result}"
    except Exception as e:
        return f"计算失败: {e}"


@tool
def get_exchange_rate(currency: str) -> str:
    """查询1单位外币兑换人民币的汇率，支持 USD、EUR、JPY"""
    rates = {"USD": 7.25, "EUR": 7.89, "JPY": 0.048}
    rate = rates.get(currency.upper())
    if rate is None:
        return f"不支持 {currency}，可用：USD、EUR、JPY"
    return f"1 {currency} = {rate} CNY"


tools = [search_weather, calculate, get_exchange_rate]

# ReAct prompt：LangChain 解析器依赖 Action/Final Answer 这两个英文关键词
REACT_TEMPLATE = """你是一个智能助手，能使用以下工具回答用户问题：

{tools}

必须严格按照以下格式输出（关键词不能改变）：

Question: 用户的问题
Thought: 分析下一步该做什么
Action: 要使用的工具名，必须是 [{tool_names}] 之一
Action Input: 传给工具的参数
Observation: 工具返回的结果
...（上面4行可以重复多次）
Thought: 我现在知道最终答案了
Final Answer: 对用户问题的完整回答

开始！

Question: {input}
Thought:{agent_scratchpad}"""

prompt = PromptTemplate.from_template(REACT_TEMPLATE)
agent = create_react_agent(llm, tools, prompt)
agent_executor = AgentExecutor(
    agent=agent,
    tools=tools,
    verbose=True,   # 打印完整推理过程
    max_iterations=6,
    handle_parsing_errors=True,
)


# ────────────────────────────────────────────
# Demo 1: 单步任务
# LLM 决定调一个工具就能回答
# ────────────────────────────────────────────
def demo_single_step():

    print("=" * 55)
    result = agent_executor.invoke({"input": "北京今天天气怎么样？"})
    print(f"\n最终回答: {result['output']}")


# ────────────────────────────────────────────
# Demo 2: 多步任务
# LLM 需要依次调用多个工具，自主规划顺序
# ────────────────────────────────────────────
def demo_multi_step():

    print("Demo 2: 多步任务（观察 Thought→Action→Observation 循环）")
    print("=" * 55)
    result = agent_executor.invoke({
        "input": "北京今天气温多少度？100美元能换多少人民币？把这两个数字加起来。"
    })
    print(f"\n最终回答: {result['output']}")


if __name__ == "__main__":
    demo_single_step()
    demo_multi_step()

## ReAct 是什么

**ReAct = Reason + Act**，是一种让 LLM 交替"思考"和"行动"的模式：

```
Thought  → 分析当前状态，决定下一步
Action   → 调用工具
Observation → 得到工具结果
Thought  → 基于结果继续分析
...
Final Answer → 得出最终答案
```

这个循环由 LangChain 的 `AgentExecutor` 自动处理：
1. 把 LLM 输出解析成 Action
2. 执行对应工具
3. 把 Observation 追加到上下文
4. 重复，直到 LLM 输出 `Final Answer`

    ...

## ReAct prompt 的关键格式

LangChain 的解析器会扫描 LLM 输出中的固定关键词：

```
Action: search_weather          ← 必须是工具名
Action Input: 北京              ← 传给工具的参数
...
Final Answer: 北京今天晴...     ← 这行出现时 Agent 停止循环
```

**这四个关键词（Action/Action Input/Observation/Final Answer）不能改成中文**，否则解析器识别不了。

## Agent vs 普通 LLM 调用的区别

| | 普通调用 | ReAct Agent |
|---|---|---|
| 工具调用次数 | 你写死（1次或固定流程） | LLM 自主决定（动态） |
| 多步推理 | 需要你拆解任务 | LLM 自动拆解 |
| 代码复杂度 | 低 | 中（框架处理了循环） |
| Token 消耗 | 少 | 多（每轮 Observation 追加上下文） |
| 适合场景 | 流程固定的任务 | 流程不固定、需要自主判断的任务 |

@tool
def get_schema() -> str:
    """获取数据库所有表的字段结构，生成 SQL 前必须先调用此工具"""
    return """
数据库包含3张表：

products(id, name, category, price)
  - category 取值：外设 / 显示器 / 音频 / 配件 / 视频
  - price：单价（元）

customers(id, name, city, join_date)
  - join_date：格式 YYYY-MM-DD

orders(id, customer_id, product_id, quantity, total_price, order_date)
  - customer_id → customers.id
  - product_id → products.id
  - total_price：订单总金额（元）
  - order_date：格式 YYYY-MM-DD
"""


@tool
def execute_sql(query: str) -> str:
    """执行 SQL 查询并返回结果，只允许 SELECT 语句"""
    query = query.strip()
    if not query.upper().startswith("SELECT"):
        return "错误：只允许 SELECT 查询，不允许修改数据"
    try:
        conn = sqlite3.connect(DB_PATH)
        c = conn.cursor()
        c.execute(query)
        rows = c.fetchall()
        cols = [d[0] for d in c.description]
        conn.close()
        if not rows:
            return "查询结果为空"
        header = " | ".join(cols)
        lines = [header, "-" * len(header)] + [" | ".join(str(v) for v in row) for row in rows]
        return "\n".join(lines)
    except Exception as e:
        return f"SQL 执行错误: {e}"

## 面试技术要点

### 为什么用 Agent 而不是直接让 LLM 写 SQL？

直接问 LLM "写一个查销售额最高产品的 SQL"，LLM 可能会：
- 猜测不存在的字段名
- 假设错误的表关系

用 Agent + get_schema 工具，LLM 先"看"了真实的表结构，再写 SQL，准确率高很多。而且如果 SQL 执行出错，Agent 能看到错误信息并自动修正。

### 为什么 execute_sql 只允许 SELECT？

防止 LLM 生成误操作：如果用户问"删掉所有订单"，没有保护的 Agent 可能真的执行 DELETE。限制为只读 SELECT，把写操作的风险完全隔离。

### 生产中还需要什么？

- **连接池**：SQLite 单连接，生产用 PostgreSQL + 连接池
- **SQL 注入防护**：用参数化查询而非字符串拼接
- **结果长度限制**：查询返回 10000 行时需要截断，避免超出 LLM context
- **权限控制**：不同用户只能查自己部门的数据
